# Huffman Encoding using Greedy Strategy

import heapq

# Create a node for Huffman Tree
class Node:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

    # Compare nodes based on frequency
    def __lt__(self, other):
        return self.freq < other.freq


# Generate Huffman Codes
def huffman_encoding(characters, frequencies):

    heap = []

    # Insert all characters into the min heap
    for char, freq in zip(characters, frequencies):
        heapq.heappush(heap, Node(char, freq))

    # Build Huffman Tree
    while len(heap) > 1:

        left = heapq.heappop(heap)
        right = heapq.heappop(heap)

        new_node = Node(None, left.freq + right.freq)
        new_node.left = left
        new_node.right = right

        heapq.heappush(heap, new_node)

    # Generate codes from Huffman Tree
    root = heap[0]
    codes = {}

    def generate_codes(node, code):
        if node is None:
            return

        if node.char is not None:
            codes[node.char] = code
            return

        generate_codes(node.left, code + "0")
        generate_codes(node.right, code + "1")

    generate_codes(root, "")

    return codes


# Main Program
characters = ['A', 'B', 'C', 'D', 'E']
frequencies = [50, 25, 15, 10, 5]

codes = huffman_encoding(characters, frequencies)

print("Huffman Codes:")

for char in characters:
    print(char, ":", codes[char])
